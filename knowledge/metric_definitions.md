# 指标定义
- play_rate = view_count / impression_count
- valid_view_rate = valid_viewer_count / viewer_count
- completion_rate = complete_view_count / view_count
- engagement_user_rate = engaged_user_count / viewer_count
- profile_visit_rate = profile_visit_count / viewer_count
- follow_conversion_rate = follow_count / profile_visit_count
- new_user_ratio = new_user_count / viewer_count
- next_day_retention_rate = next_day_active_user_count / new_user_count
